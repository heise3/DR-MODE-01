            kcal <- q[[paste0(dr, "IKCAL")]]

                n_items = as.numeric(validline)), q$SEQN, na.rm = FALSE)
            agg <- data.frame(SEQN = as.numeric(rownames(agg)), 
                agg, row.names = NULL)
            iff[[tag]] <- agg

                "TOT")), ".xpt"))))
            di <- demo[match(tot$SEQN, demo$SEQN), c("RIDAGEYR", 
                "SDMVSTRA", "SDMVPSU")]
            stopifnot(!anyNA(di$RIDAGEYR), !anyDuplicated(tot$SEQN))

            rel <- tot[[paste0(dr, "DRSTZ")]] == 1

                na.rm = TRUE)
            missing_fast <- sum(rel & denom > 0 & is.na(a$fast_kcal), 
                na.rm = TRUE)
            stopifnot(missing_rows == 0, missing_fast == 0)
            tot$share <- 100 * a$fast_kcal/denom
            tot$share[!is.finite(tot$share) | is.na(denom) | 
                denom <= 0] <- NA_real_
            dif <- a$iff_kcal - denom
            sel <- rel & denom > 0 & is.finite(dif)
            qa[[tag]] <- data.frame(cycle = cy, day = day, iff_rows = nrow(q), 
                iff_persons = nrow(agg), duplicate_food_keys = duplicate, 
                reliable_positive_energy_missing_IFF = missing_rows, 
                missing_fast_amount = missing_fast, max_abs_energy_sum_minus_TOT = max(abs(dif[sel])), 
                n_nonzero_energy_sum_minus_TOT = sum(abs(dif[sel]) > 
                  1e-08), n_share_above_100 = sum(tot$share > 
                  100, na.rm = TRUE), n_share_below_0 = sum(tot$share < 
                  0, na.rm = TRUE), reliable_adult_zero_energy = sum(rel & 
                  tot$RIDAGEYR >= 20 & denom == 0, na.rm = TRUE))
            tod[[tag]] <- tot
        }

        L = 11.7), tn <- c(H = 5045, I = 5015, P = 7705, L = 4792), 
    for (cy in names(targets)) {

        wv <- if (cy == "P") 
            "WTDRD1PP"
        else "WTDRD1"
        d$w <- d[[wv]]
        d <- d[!is.na(d$w) & d$w > 0, ]

            0 & is.finite(d$share)

            weights = ~w, data = d, nest = TRUE)
        ds <- subset(des, eligible)
        m <- svymean(~share, ds)
        e <- as.numeric(coef(m))
        s <- as.numeric(SE(m))
        df <- degf(ds)
        ci <- e + c(-1, 1) * qt(0.975, df) * s
        cal[[cy]] <- data.frame(cycle = cy, n_persons = sum(d$eligible), 
            official_n = tn[cy], mean_pct = e, se = s, df = df, 
            ci_low = ci[1], ci_high = ci[2], official_mean_pct = targets[cy], 
            error_pp = e - targets[cy], absolute_error_pp = abs(e - 
                targets[cy]), status = if (abs(e - targets[cy]) <= 
                0.3) 
                "PASS"
            else "FAIL")
    }, cal <- do.call(rbind, cal), outcsv(cal, "results/calibration.csv"), 
    fg <- data.frame(cycle = cal$cycle, time = c(H = 2014, I = 2016, 
        P = 2018.6, L = 2022.7)[cal$cycle], scenario = "original", 
        mean_pct = cal$mean_pct, se = cal$se, ci_low = cal$ci_low, 
        ci_high = cal$ci_high, n_persons = cal$n_persons, inference = "Taylor", 
        status = "CALIBRATION_ONLY", correction_status = "NOT_RUN", 
        trend_status = "NOT_RUN"), outcsv(fg, "figdata/x1_trend.csv"), 
    run$calibration <- if (all(cal$status == "PASS")) "PASS" else "FAIL", 
    run$stage <- "CALIBRATION", if (run$calibration == "FAIL") {
        run$verdict <- "CALIBRATION_FAIL"
        outjson(run, "results/run_status.json")
        quit(status = 0)
    }, outjson(run, "results/run_status.json"), print(cal, row.names = FALSE), 
    z <- readRDS("outputs/S1_writing/provenance/analysis_input.rds"), 
    d <- z$data, for (day in 1:2) {
        keys <- paste(d$cycle, d$SEQN, sep = ":")
        a <- do.call(rbind, lapply(cycs, function(cy) {
            x <- iff[[paste(cy, day, sep = "_")]]
            x$key <- paste(cy, x$SEQN, sep = ":")
            x
        }))
        den <- d[[paste0("DR", day, "TKCAL")]]
        v <- 100 * a$fast_kcal[match(keys, a$key)]/den
        v[!is.finite(v) | is.na(den) | den <= 0] <- NA_real_
        d[[paste0("share", day)]] <- v
    }, d$D <- d$share2 - d$share1, d$outcome_complete <- is.finite(d$D), 
    raw <- readRDS("work/nhanes_stage_a_merged.rds"), flow <- do.call(rbind, 
        lapply(cycs, function(cy) {
            x <- d[d$cycle == cy, ]
            data.frame(cycle = cy, frozen_eligible_persons = sum(raw[[cy]]$adult), 
                positive_weight_adults = sum(x$adult), paired_ratio_defined = sum(x$adult & 
                  x$outcome_complete), covariates_complete = sum(x$adult & 
                  x$cov_complete), model_complete = sum(x$adult & 
                  x$cov_complete & x$outcome_complete))
        })), outcsv(flow, "results/person_counts.csv"), dc <- d[d$cycle != 
        "L", ], dc$cycle <- droplevels(dc$cycle), dc$w <- dc$WTDR2D/4, 
    cd <- svydesign(ids = ~psu, strata = ~stratum, weights = ~w, 
        data = dc, nest = TRUE), cd <- subset(cd, adult & outcome_complete), 
    hm <- svyglm(D ~ cycle, design = cd), ht <- regTermTest(hm, 
        ~cycle, method = "Wald"), tr <- svyglm(D ~ time, design = cd), 
    a <- c(1, 2022.7 - 2015), ex <- sum(a * coef(tr)), xs <- sqrt(drop(t(a) %*%

        n_persons = nobs(hm), wald_F = as.numeric(ht$Ftest), 
        wald_num_df = as.numeric(ht$df), wald_den_df = as.numeric(ht$ddf), 
        wald_p = hp, control_D_mean = as.numeric(coef(mu)), control_D_se = as.numeric(SE(mu)), 
        trend_per_year = unname(coef(tr)["time"]), trend_se = unname(SE(tr)["time"]), 
        trend_df = tr$df.residual, trend_p = summary(tr)$coefficients["time", 
            4], L_extrapolated_D = ex, L_extrapolated_se = xs,
